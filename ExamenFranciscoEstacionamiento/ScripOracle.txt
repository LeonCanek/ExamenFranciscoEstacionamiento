CREATE TABLE Estancia (
    id NUMBER PRIMARY KEY,
    hora_entrada TIMESTAMP NOT NULL,
    hora_salida TIMESTAMP,
    vehiculo_placa VARCHAR2(20) NOT NULL,
    FOREIGN KEY (vehiculo_placa) REFERENCES Vehiculo(placa)
);
